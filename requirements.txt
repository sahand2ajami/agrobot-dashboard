# Dual-Robot Dashboard (agrobot | jackal) — Python dependencies
# Tested on Python 3.10 / Ubuntu 22.04 / ROS 2 Humble

# ─────────────────────────────────────────────────────────────────────────────

# Object detection (YOLOv8) — includes torch / torchvision as transitive deps
ultralytics>=8.4

# Computer vision — MJPEG encoding in serve.py and legacy object_detector.py
opencv-python>=4.5

# Numerical arrays — detection pipeline (also pulled in by ultralytics)
numpy>=1.21

# Serial communication — GNSS receiver (scripts/gnss_rtu608bt_read.py)
pyserial>=3.5

# Chassis YAML config (config/chassis/*.yaml). Required — the old hand-rolled
# fallback parser was removed (it silently mis-parsed anything non-trivial).
pyyaml>=6.0

# Modbus TCP — direct PLC communication (dashboard/plc_client.py, dashboard/amr_plc_serve.py).
# Imported lazily; the dashboard and tests run without a live PLC.
pymodbus>=3.0

# Test runner
pytest>=6.0
